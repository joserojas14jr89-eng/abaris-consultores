// URL DE TU BACKEND EN GOOGLE APPS SCRIPT (CONECTADA)
const API_URL = "https://script.google.com/macros/s/AKfycbz3GNdBNC52cfPWHdrNFIVgOgBz9pIhafOcK7dJsyXmUSksOVY27DattCK-EIkd0rSz/exec";

const formatCurrency = (amount) => {
  return new Intl.NumberFormat('es-PY', { style: 'currency', currency: 'PYG' }).format(Number(amount) || 0);
};

const AbarisLogo = ({ className = "w-8 h-8" }) => (
  <svg viewBox="0 0 100 100" fill="none" xmlns="http://www.w3.org/2000/svg" className={className}>
    <path d="M50 15 L20 80 L32 80 L50 40 L68 80 L80 80 Z" fill="#C5A059" />

export default function AbarisApp() {
  const [viewMode, setViewMode] = useState('celular'); // 'celular' o 'escritorio'
  const [mockEmail, setMockEmail] = useState('');
  
  // Estados de datos
  const [invoices, setInvoices] = useState([]);
  const [usersData, setUsersData] = useState([]);
  const [rucsDirectory, setRucsDirectory] = useState({});
  const [cobranzas, setCobranzas] = useState([]);
  const [liquidaciones, setLiquidaciones] = useState([]);
  const [abarisGastos, setAbarisGastos] = useState([]);
  const [mesesEstados, setMesesEstados] = useState({});
  const [isLoading, setIsLoading] = useState(false);

  // Sincronizar con Google Sheets al iniciar
  useEffect(() => {
    cargarDatosReales();
  }, []);

  const cargarDatosReales = async () => {
    setIsLoading(true);
    try {
      const res = await fetch(API_URL);
      const json = await res.json();
      if (json.status === "success" && json.data) {
        setInvoices(json.data.invoices || []);
        setUsersData(json.data.usersData || []);
        setCobranzas(json.data.cobranzas || []);
        setLiquidaciones(json.data.liquidaciones || []);
        setAbarisGastos(json.data.abarisGastos || []);
        
        // Mapear estados de meses
        let estadosMap = {};
        (json.data.mesesEstados || []).forEach(m => {
          estadosMap[`${m.email}-${m.mes}`] = m.estado;
        });
        setMesesEstados(estadosMap);

        // Mapear directorio de RUCs
        let rucsMap = {};
        (json.data.rucsDirectory || []).forEach(r => {
          rucsMap[r.ruc] = r.razonSocial;
        });
        setRucsDirectory(rucsMap);
      }
    } catch (e) {
      console.error("Error al sincronizar con Google Sheets:", e);
    }
    setIsLoading(false);
  };

  const enviarAlBackend = async (action, payload) => {
    try {
      await fetch(API_URL, {
        method: "POST",
        headers: { "Content-Type": "text/plain;charset=utf-8" },
        body: JSON.stringify({ action, payload })
      });
      await cargarDatosReales(); // Recargar datos frescos
    } catch (e) {
      console.error("Error al enviar acción:", e);
    }
  };

  const db = {
    addInvoice: (data) => enviarAlBackend("addInvoice", data),
    addUser: (data) => enviarAlBackend("addUser", data),
    updateUser: (id, data) => enviarAlBackend("updateUser", { id, data }),
    addCobranza: (data) => enviarAlBackend("addCobranza", data),
    updateCobranza: (id, data) => enviarAlBackend("updateCobranza", { id, data }),
    addLiquidacion: (data) => enviarAlBackend("addLiquidacion", data),
    addAbarisGasto: (data) => enviarAlBackend("addAbarisGasto", data),
    setMesEstado: (email, mes, estado) => enviarAlBackend("setMesEstado", { email, mes, estado }),
    addRuc: (ruc, razonSocial) => enviarAlBackend("addRuc", { ruc, razonSocial })
  };

  const [loginInput, setLoginInput] = useState('');
  const handleLogin = (e) => {
    e.preventDefault();
    if (!loginInput) return;
    if (loginInput !== 'admin@abaris.com' && !usersData.find(u => u.email === loginInput)) {
      db.addUser({ email: loginInput, razonSocial: 'Nuevo Cliente', ruc: '8000000-0', telefono: '0981000000', honorarios: 300000 });
    }
    setMockEmail(loginInput);
  };

  const quickLogin = (email, defaultName) => {
    if (email !== 'admin@abaris.com' && !usersData.find(u => u.email === email)) {
      db.addUser({ email, razonSocial: defaultName, ruc: '80012345-6', telefono: '0981000000', honorarios: 300000 });
    }
    setMockEmail(email);
  };

  if (!mockEmail) {
    return (
      <div className="min-h-screen flex items-center justify-center p-4" style={{ backgroundColor: '#0B132B' }}>
        <div className="bg-white p-8 rounded-2xl shadow-2xl max-w-md w-full text-center border-t-8" style={{ borderColor: '#C5A059' }}>
          <div className="flex justify-center mb-4"><AbarisLogo className="w-20 h-20" /></div>
          <h1 className="text-3xl font-bold text-slate-900 mb-1">ABARIS</h1>
          <h2 className="text-sm font-semibold tracking-widest uppercase mb-8" style={{ color: '#C5A059' }}>Consultores</h2>
          <form onSubmit={handleLogin} className="space-y-4">
            <input type="email" placeholder="Ingresa tu correo electrónico" className="w-full p-3 border rounded-lg outline-none" value={loginInput} onChange={e => setLoginInput(e.target.value)} required />
            <button type="submit" className="w-full text-white font-semibold p-3 rounded-lg shadow-md" style={{ backgroundColor: '#C5A059' }}>Ingresar al Portal</button>
          </form>
          <div className="mt-8 border-t pt-6">
            <p className="text-xs font-semibold text-gray-400 mb-4 uppercase">Accesos de Prueba</p>
            <div className="flex flex-col gap-3">
              <button onClick={() => quickLogin('admin@abaris.com')} className="w-full text-white text-sm p-3 rounded-lg flex items-center justify-center shadow-sm" style={{ backgroundColor: '#0B132B' }}><Shield size={16} className="mr-2"/> Entrar como Administrador</button>
              <button onClick={() => quickLogin('cliente1@empresa.com', 'Comercial El Sol S.R.L.')} className="w-full text-sm p-3 rounded-lg flex items-center justify-center border transition-colors" style={{ backgroundColor: '#F8F9FA', color: '#6B4B9A', borderColor: '#6B4B9A' }}><Users size={16} className="mr-2"/> Entrar como Cliente</button>
            </div>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-100 flex flex-col">
      {/* Barra de Control de Vista */}
      <div className="bg-slate-900 text-white px-4 py-2 flex justify-between items-center text-xs">
        <span className="flex items-center"><Shield size={14} className="mr-1 text-amber-400"/> Conectado a Google Sheets & Drive</span>
        <div className="flex gap-2">
          <button onClick={() => setViewMode('celular')} className={`px-3 py-1 rounded ${viewMode === 'celular' ? 'bg-amber-600 font-bold' : 'bg-slate-700'}`}>Vista Celular (Cliente)</button>
          <button onClick={() => setViewMode('escritorio')} className={`px-3 py-1 rounded ${viewMode === 'escritorio' ? 'bg-amber-600 font-bold' : 'bg-slate-700'}`}>Vista Escritorio (Admin)</button>
        </div>
      </div>

      <div className="flex-1 flex justify-center items-start p-0 md:p-4">
        <div className={`w-full ${viewMode === 'celular' ? 'max-w-md shadow-2xl rounded-3xl overflow-hidden border-8 border-slate-900 bg-white min-h-[800px]' : 'max-w-7xl bg-white rounded-xl shadow-lg'}`}>
          {mockEmail === 'admin@abaris.com' ? (
            <AdminDashboard invoices={invoices} usersData={usersData} cobranzas={cobranzas} liquidaciones={liquidaciones} abarisGastos={abarisGastos} mesesEstados={mesesEstados} onLogout={() => setMockEmail('')} db={db} />
          ) : (
            <ClientDashboard mockEmail={mockEmail} invoices={invoices.filter(i => i.userEmail === mockEmail)} cobranzas={cobranzas.filter(c => c.userEmail === mockEmail)} liquidaciones={liquidaciones.filter(l => l.userEmail === mockEmail)} rucsDirectory={rucsDirectory} userData={usersData.find(u => u.email === mockEmail)} mesesEstados={mesesEstados} onLogout={() => setMockEmail('')} db={db} />
          )}
        </div>
      </div>
      {isLoading && <div className="fixed bottom-4 right-4 bg-slate-900 text-white px-4 py-2 rounded-full text-xs shadow-lg flex items-center"><Clock size={14} className="animate-spin mr-2"/> Sincronizando con Google...</div>}
    </div>
  );
}

function AdminFacturas({ invoices, clients, db, mesesEstados }) {
  const [importClient, setImportClient] = useState('');
  const [importPeriod, setImportPeriod] = useState('');
  const [exportClient, setExportClient] = useState('');
  const [exportPeriod, setExportPeriod] = useState('');

  const handleImportMarangatu = (e) => {
    e.preventDefault();
    if (!importClient || !importPeriod) return alert('Selecciona cliente y periodo.');
    db.addInvoice({
      userEmail: importClient,
      fecha: `${importPeriod}-15`,
      ruc: '80000000',
      razonSocial: 'PROVEEDOR ELECTRÓNICO S.A. (MARANGATÚ)',
      tipo: 'Gasto',
      monto: 850000,
      iva10: 77272,
      cdc: '05800261577001002036773922026101314320998022',
      origen: 'ELECTRONICO'
    });
    db.setMesEstado(importClient, importPeriod, 'Pendiente');
    alert('¡Excel de Marangatú procesado! El balance se actualizó y el cliente debe aprobarlo.');
  };

  const handleExportSet = (e) => {
    e.preventDefault();
    if (!exportClient || !exportPeriod) return alert('Selecciona cliente y periodo.');
    const estado = mesesEstados[`${exportClient}-${exportPeriod}`];
    if (estado !== 'Aprobado') return alert('No puedes exportar: El cliente aún no ha dado su Aprobación (OK) para este periodo.');

    // Filtrar comprobantes del cliente y periodo que NO sean electrónicos (sin CDC) para el archivo SET oficial
    const comprobantesFisicos = invoices.filter(inv => inv.userEmail === exportClient && inv.fecha && inv.fecha.startsWith(exportPeriod) && !inv.cdc);

    // Generar formato SET (Especificación Técnica Junio 2021: delimitado por punto y coma)
    let csvContent = "";
    comprobantesFisicos.forEach(inv => {
      const tipoReg = inv.tipo === 'Ingreso' ? '1' : '2';
      const tipoId = '11';
      const rucComprador = inv.ruc || '80000000';
      const razonSoc = inv.razonSocial || 'SIN NOMBRE';
      const tipoComp = '109';
      const fechaFmt = inv.fecha.split('-').reverse().join('/');
      const timbrado = '12345678';
      const nroComp = '001-001-0000001';
      const grav10 = inv.iva10 ? inv.monto : 0;
      const grav5 = inv.iva5 ? inv.monto : 0;
      const exento = 0;
      const total = inv.monto;
      const condicion = '1';
      const monedaExt = 'N';
      const imputaIva = 'S';
      const imputaIre = 'N';
      const imputaIrp = 'S';
      const noImputa = 'N';

      csvContent += `${tipoReg};${tipoId};${rucComprador};${razonSoc};${tipoComp};${fechaFmt};${timbrado};${nroComp};${grav10};${grav5};${exento};${total};${condicion};${monedaExt};${imputaIva};${imputaIre};${imputaIrp};${noImputa}\n`;
    });

    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.setAttribute("href", url);
    link.setAttribute("download", `80000000_REG_${exportPeriod.replace('-','')}_V0001.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    alert('¡Archivo oficial SET (.CSV) generado con éxito según la normativa!');
  };

  return (
    <div className="space-y-6">
      <div className="bg-white rounded-xl shadow-sm border p-6">
        <h3 className="text-lg font-bold mb-2 flex items-center text-slate-800"><Download className="mr-2 text-purple-700"/> 1. Importar Facturas Electrónicas (Excel Marangatú)</h3>
        <form onSubmit={handleImportMarangatu} className="flex flex-col md:flex-row gap-4 items-end mt-4">
          <div className="w-full md:w-1/3">
            <label className="block text-xs font-semibold mb-1">Cliente</label>
            <select required className="w-full border p-2 rounded-lg text-sm" value={importClient} onChange={e => setImportClient(e.target.value)}>
              <option value="">Seleccionar...</option>
              {clients.map(c => <option key={c.email} value={c.email}>{c.razonSocial || c.email}</option>)}
            </select>
          </div>
          <div className="w-full md:w-1/4">
            <label className="block text-xs font-semibold mb-1">Periodo</label>
            <input required type="month" className="w-full border p-2 rounded-lg text-sm" value={importPeriod} onChange={e => setImportPeriod(e.target.value)} />
          </div>
          <div className="w-full md:w-1/4">
            <label className="block text-xs font-semibold mb-1">Archivo Excel Marangatú</label>
            <input required type="file" accept=".xlsx,.csv" className="w-full text-sm border p-1 rounded-lg" />
          </div>
          <button type="submit" className="text-white px-5 py-2.5 rounded-lg text-sm font-semibold bg-purple-700">Importar Excel</button>
        </form>
      </div>

      <div className="bg-white rounded-xl shadow-sm border p-6">
        <h3 className="text-lg font-bold mb-2 flex items-center text-slate-800"><UploadCloud className="mr-2 text-amber-600"/> 2. Exportar Archivo SET Oficial (.CSV)</h3>
        <form onSubmit={handleExportSet} className="flex flex-col md:flex-row gap-4 items-end mt-4">
          <div className="w-full md:w-1/3">
            <label className="block text-xs font-semibold mb-1">Cliente</label>
            <select required className="w-full border p-2 rounded-lg text-sm" value={exportClient} onChange={e => setExportClient(e.target.value)}>
              <option value="">Seleccionar...</option>
              {clients.map(c => <option key={c.email} value={c.email}>{c.razonSocial || c.email}</option>)}
            </select>
          </div>
          <div className="w-full md:w-1/4">
            <label className="block text-xs font-semibold mb-1">Periodo a Exportar</label>
            <input required type="month" className="w-full border p-2 rounded-lg text-sm" value={exportPeriod} onChange={e => setExportPeriod(e.target.value)} />
          </div>
          <button type="submit" className="text-white px-5 py-2.5 rounded-lg text-sm font-semibold bg-slate-900 flex items-center"><Download className="mr-2" size={16}/> Descargar .CSV SET</button>
        </form>
      </div>
    </div>
  );
}

function ClientEditCard({ client, db }) {
  const [isEditing, setIsEditing] = useState(false);
  const [data, setData] = useState(client);
  const [showPassword, setShowPassword] = useState(false);

  const handleSave = () => {
    db.updateUser(client.id || client.email, data);
    setIsEditing(false);
  };

  return (
    <div className="border p-4 rounded-xl bg-slate-50">
      {!isEditing ? (
        <div className="flex flex-col md:flex-row justify-between items-start md:items-center">
          <div>
            <h4 className="font-bold text-slate-800 text-lg">{client.razonSocial || 'Sin Nombre'}</h4>
            <p className="text-sm text-slate-500">{client.email} | RUC: {client.ruc || 'N/A'}</p>
            <div className="flex gap-4 mt-2 text-xs font-semibold text-slate-600">
              <span>Tel: {client.telefono || 'N/A'}</span>
              <span className="text-amber-600">Honorarios: {formatCurrency(client.honorarios)}</span>
              <span className="flex items-center text-purple-700">
                Marangatú Pwd: {showPassword ? (client.marangatuPassword || 'No cargada') : '••••••••'}
                <button onClick={() => setShowPassword(!showPassword)} className="ml-2 text-slate-500 hover:text-slate-800"><Eye size={14}/></button>
              </span>
            </div>
          </div>
          <button onClick={() => setIsEditing(true)} className="mt-4 md:mt-0 flex items-center text-sm border px-4 py-2 rounded-lg bg-white shadow-sm"><Edit2 size={14} className="mr-2"/> Editar Datos</button>
        </div>
      ) : (
        <div className="space-y-3">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
            <input type="text" placeholder="Razón Social" className="border p-2 rounded text-sm bg-white" value={data.razonSocial || ''} onChange={e => setData({...data, razonSocial: e.target.value})} />
            <input type="text" placeholder="RUC" className="border p-2 rounded text-sm bg-white" value={data.ruc || ''} onChange={e => setData({...data, ruc: e.target.value})} />
            <input type="text" placeholder="Teléfono" className="border p-2 rounded text-sm bg-white" value={data.telefono || ''} onChange={e => setData({...data, telefono: e.target.value})} />
            <input type="number" placeholder="Honorarios Gs." className="border p-2 rounded text-sm bg-white" value={data.honorarios || 0} onChange={e => setData({...data, honorarios: e.target.value})} />
            <input type="text" placeholder="Contraseña Marangatú" className="border p-2 rounded text-sm bg-white md:col-span-2" value={data.marangatuPassword || ''} onChange={e => setData({...data, marangatuPassword: e.target.value})} />
          </div>
          <div className="flex justify-end gap-2">
            <button onClick={() => setIsEditing(false)} className="px-3 py-1 text-slate-600 text-sm">Cancelar</button>
            <button onClick={handleSave} className="px-4 py-1 bg-slate-900 text-white rounded text-sm">Guardar</button>
          </div>
        </div>
      )}
    </div>
  );
}

function AdminCobranzaForm({ clients, db }) {
  const [form, setForm] = useState({ userEmail: '', periodo: 'Noviembre 2026', monto: '' });
  const handleSubmit = (e) => {
    e.preventDefault();
    db.addCobranza({ ...form, monto: Number(form.monto), estado: 'Pendiente', createdAt: new Date().toISOString() });
    setForm({ userEmail: '', periodo: 'Noviembre 2026', monto: '' });
    alert('Cobro de honorarios generado con éxito para el cliente.');
  };
  return (
    <form onSubmit={handleSubmit} className="border p-4 rounded-xl bg-slate-50 mb-6">
      <h4 className="font-semibold mb-3 text-sm text-slate-700">Generar Cobro de Honorarios</h4>
      <div className="grid grid-cols-1 md:grid-cols-4 gap-3">
        <select required className="border p-2 rounded text-sm bg-white" value={form.userEmail} onChange={e => setForm({...form, userEmail: e.target.value})}>
          <option value="">Seleccionar Cliente...</option>
          {clients.map(c => <option key={c.email} value={c.email}>{c.razonSocial || c.email}</option>)}
        </select>
        <input required type="text" placeholder="Periodo" className="border p-2 rounded text-sm bg-white" value={form.periodo} onChange={e => setForm({...form, periodo: e.target.value})} />
        <input required type="number" placeholder="Monto Gs." className="border p-2 rounded text-sm bg-white" value={form.monto} onChange={e => setForm({...form, monto: e.target.value})} />
        <button type="submit" className="bg-amber-600 text-white p-2 rounded text-sm font-semibold flex items-center justify-center"><PlusCircle size={16} className="mr-1"/> Generar Deuda</button>
      </div>
    </form>
  );

function ClientInicio({ invoices, mockEmail, mesesEstados, db }) {
  const [filterMonth, setFilterMonth] = useState('2026-11');
  const [formData, setFormData] = useState({ ruc: '', razonSocial: '', fecha: new Date().toISOString().split('T')[0], tipo: 'Gasto', monto: '', iva10: '' });
  const [isUploadingCard, setIsUploadingCard] = useState(false);

  const estadoMes = mesesEstados[`${mockEmail}-${filterMonth}`] || 'Pendiente';

  const summary = useMemo(() => {
    const filtered = invoices.filter(inv => inv.fecha && inv.fecha.startsWith(filterMonth));
    const ingresos = filtered.filter(i => i.tipo === 'Ingreso').reduce((acc, curr) => acc + Number(curr.monto), 0);
    const gastos = filtered.filter(i => i.tipo === 'Gasto').reduce((acc, curr) => acc + Number(curr.monto), 0);
    const ivaVentas = ingresos / 11;
    const ivaCompras = gastos / 11;
    const saldoIva = ivaVentas - ivaCompras;
    return { ingresos, gastos, saldoIva };
  }, [invoices, filterMonth]);

  const handleSaveInvoice = (e) => {
    e.preventDefault();
    db.addInvoice({
      userEmail: mockEmail,
      fecha: formData.fecha,
      ruc: formData.ruc,
      razonSocial: formData.razonSocial,
      tipo: formData.tipo,
      monto: Number(formData.monto),
      iva10: formData.tipo === 'Gasto' ? Math.round(Number(formData.monto) / 11) : 0,
      origen: 'VIRTUAL'
    });
    setFormData({ ruc: '', razonSocial: '', fecha: new Date().toISOString().split('T')[0], tipo: 'Gasto', monto: '', iva10: '' });
    alert('¡Factura guardada con éxito!');
  };

  const handleUploadCardStatement = (e) => {
    const file = e.target.files[0];
    if (!file) return;
    setIsUploadingCard(true);
    const reader = new FileReader();
    reader.onload = () => {
      db.addInvoice({
        userEmail: mockEmail,
        fecha: `${filterMonth}-10`,
        ruc: '80000000',
        razonSocial: 'EXTRACTO TARJETA (SERVICIO DIGITAL)',
        tipo: 'Gasto',
        monto: 150000,
        iva10: 13636,
        fileBase64: reader.result.split(',')[1],
        fileName: file.name,
        mimeType: file.type,
        origen: 'EXTRACTO'
      });
      setIsUploadingCard(false);
      alert('¡Extracto de tarjeta subido con éxito a Google Drive para cálculo de IVA Ley!');
    };
    reader.readAsDataURL(file);
  };

  return (
    <div className="space-y-4">
      {/* Banner Recordatorio Servicios Digitales */}
      <div className="bg-amber-50 border border-amber-300 p-4 rounded-2xl shadow-sm text-amber-900">
        <div className="flex items-start gap-3">
          <Bell className="text-amber-600 shrink-0 mt-1" size={20}/>
          <div>
            <h4 className="font-bold text-sm">Recordatorio de Gastos Digitales</h4>
            <p className="text-xs mt-0.5 text-amber-800">Si tuviste consumos en servicios digitales (Netflix, Spotify, Airbnb, etc.), sube tu extracto de tarjeta del mes para computar el IVA Ley.</p>
            <label className="mt-3 inline-flex items-center bg-amber-600 text-white px-3 py-1.5 rounded-lg text-xs font-bold cursor-pointer hover:bg-amber-700">
              <UploadCloud size={14} className="mr-1.5"/> {isUploadingCard ? 'Subiendo...' : 'Subir Extracto Tarjeta'}
              <input type="file" className="hidden" onChange={handleUploadCardStatement} />
            </label>
          </div>
        </div>
      </div>

      {/* Selector de Periodo y Estado */}
      <div className="bg-white p-4 rounded-2xl shadow-sm border flex justify-between items-center">
        <div>
          <span className="text-[10px] font-bold text-slate-400 uppercase">Periodo Activo</span>
          <select className="block font-bold text-slate-800 text-sm outline-none bg-transparent" value={filterMonth} onChange={e => setFilterMonth(e.target.value)}>
            <option value="2026-11">Noviembre 2026</option>
            <option value="2026-12">Diciembre 2026</option>
          </select>
        </div>
        <div>
          {estadoMes === 'Aprobado' ? (
            <span className="bg-emerald-100 text-emerald-800 text-xs px-3 py-1 rounded-full font-bold flex items-center"><CheckCircle size={12} className="mr-1"/> Aprobado por Cliente</span>
          ) : (
            <button onClick={() => db.setMesEstado(mockEmail, filterMonth, 'Aprobado')} className="bg-slate-900 text-white text-xs px-4 py-2 rounded-xl font-bold shadow-sm">Aprobar Mes (OK)</button>
          )}
        </div>
      </div>

      {/* Tarjetas Resumen */}
      <div className="grid grid-cols-2 gap-3">
        <div className="bg-white p-4 rounded-2xl border shadow-sm">
          <span className="text-[10px] font-bold text-slate-400 uppercase">Total Ingresos</span>
          <p className="text-lg font-black text-slate-800 mt-1">{formatCurrency(summary.ingresos)}</p>
        </div>
        <div className="bg-white p-4 rounded-2xl border shadow-sm">
          <span className="text-[10px] font-bold text-slate-400 uppercase">Total Gastos</span>
          <p className="text-lg font-black text-slate-800 mt-1">{formatCurrency(summary.gastos)}</p>
        </div>
        <div className="bg-white p-4 rounded-2xl border shadow-sm col-span-2 flex justify-between items-center" style={{ backgroundColor: summary.saldoIva >= 0 ? '#0B132B' : '#6B4B9A', color: 'white' }}>
          <div>
            <span className="text-[10px] font-bold text-amber-300 uppercase">{summary.saldoIva >= 0 ? 'IVA Estimado a Pagar' : 'IVA Crédito a Favor'}</span>
            <p className="text-xl font-black mt-1">{formatCurrency(Math.abs(summary.saldoIva))}</p>
          </div>
          <Receipt size={28} className="text-amber-300 opacity-80"/>
        </div>
      </div>

      {/* Carga Manual de Factura */}
      <div className="bg-white p-4 rounded-2xl border shadow-sm space-y-3">
        <h3 className="font-bold text-sm text-slate-800 flex items-center"><Camera className="mr-1.5 text-purple-700" size={16}/> Cargar Comprobante / Factura</h3>
        <form onSubmit={handleSaveInvoice} className="space-y-3 text-xs">
          <div className="grid grid-cols-2 gap-2">
            <input required type="text" placeholder="RUC Proveedor" className="border p-2 rounded-lg bg-slate-50" value={formData.ruc} onChange={e => setFormData({...formData, ruc: e.target.value})} />
            <input required type="text" placeholder="Razón Social" className="border p-2 rounded-lg bg-slate-50" value={formData.razonSocial} onChange={e => setFormData({...formData, razonSocial: e.target.value})} />
          </div>
          <div className="grid grid-cols-2 gap-2">
            <select className="border p-2 rounded-lg bg-slate-50" value={formData.tipo} onChange={e => setFormData({...formData, tipo: e.target.value})}>
              <option value="Gasto">Compra / Gasto</option>
              <option value="Ingreso">Venta / Ingreso</option>
            </select>
            <input required type="date" className="border p-2 rounded-lg bg-slate-50" value={formData.fecha} onChange={e => setFormData({...formData, fecha: e.target.value})} />
          </div>
          <input required type="number" placeholder="Monto Total Gs." className="w-full border p-2 rounded-lg bg-slate-50 font-bold" value={formData.monto} onChange={e => setFormData({...formData, monto: e.target.value})} />
          <button type="submit" className="w-full bg-slate-900 text-white py-2.5 rounded-xl font-bold">Guardar Factura</button>
        </form>
      </div>
    </div>
  );
}

          <p><strong className="text-slate-400">Cuenta N°:</strong> <span className="font-mono font-bold text-purple-700">13351759001</span></p>
        </div>
      </div>

      <div className="bg-white p-4 rounded-2xl border shadow-sm space-y-3">
        <h3 className="font-bold text-sm mb-3 flex items-center text-slate-800"><CreditCard className="mr-1.5 text-purple-700" size={16}/> Mis Honorarios</h3>
